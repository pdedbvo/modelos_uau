rules_version = '2';
service cloud.firestore {
  match /databases/{db}/documents {
    function logado() { return request.auth != null && request.auth.token.email != null; }
    function dono()   { return logado() && request.auth.token.email == 'sergio.f@edificabpo.com.br'; }
    function cad()    { return /databases/$(db)/documents/editores/$(request.auth.token.email); }
    function editor() { return dono() || (logado() && exists(cad())); }
    function admin()  { return dono() || (logado() && exists(cad()) && get(cad()).data.papel == 'Administrador'); }

    match /conteudo/{doc}   { allow read: if true;     allow write: if editor(); }
    match /editores/{email} { allow read: if editor(); allow write: if admin(); }

    // ---------- Onboarding: Administrador, RH e Líder ----------
    function acOnb()    { return /databases/$(db)/documents/acessosOnboarding/$(request.auth.token.email); }
    function temOnb()   { return logado() && exists(acOnb()); }
    function papelOnb() { return get(acOnb()).data.papel; }
    function onbAdmin() { return editor() || (temOnb() && papelOnb() == 'Administrador'); }
    function onbRH()    { return onbAdmin() || (temOnb() && papelOnb() == 'RH'); }

    match /onboarding/{id} {
      allow read, update: if onbRH() || (temOnb() && resource.data.liderEmail == request.auth.token.email);
      allow create: if onbRH() || (temOnb() && request.resource.data.liderEmail == request.auth.token.email);
      allow delete: if onbRH();
    }
    match /acessosOnboarding/{email} {
      allow read: if logado() && (request.auth.token.email == email || onbRH());
      allow write: if onbAdmin();
    }

    // ---------- Lições aprendidas ----------
    function sessao()   { return request.auth != null; }
    function licAdmin() { return editor() || (logado() && exists(/databases/$(db)/documents/licoesAdmins/$(request.auth.token.email))); }
    function autor24h() { return sessao() && resource.data.autorUid == request.auth.uid
                                 && request.time < resource.data.criadoEm + duration.value(24, 'h'); }
    function txt(d, k, max) { return !(k in d) || (d[k] is string && d[k].size() <= max); }
    function licaoOk(d) {
      return d.titulo is string && d.titulo.size() >= 3 && d.titulo.size() <= 140
        && d.problema is string && d.problema.size() >= 1 && d.problema.size() <= 4000
        && d.autor is string && d.autor.size() >= 1 && d.autor.size() <= 80
        && d.categoria is string && d.categoria.size() <= 60
        && txt(d, 'evitar', 4000) && txt(d, 'licao', 4000) && txt(d, 'natureza', 10) && txt(d, 'causa', 4000) && txt(d, 'solucao', 4000) && txt(d, 'tags', 300) && txt(d, 'link', 600)
        && txt(d, 'onde', 80) && txt(d, 'etapa', 80) && txt(d, 'quando', 7) && txt(d, 'categoriaNome', 60);
    }
    match /licoes/{id} {
      allow read: if true;
      allow create: if sessao() && request.resource.data.autorUid == request.auth.uid
                    && request.resource.data.criadoEm == request.time && licaoOk(request.resource.data);
      allow update: if (licAdmin() || autor24h()) && licaoOk(request.resource.data)
                    && request.resource.data.autorUid == resource.data.autorUid
                    && request.resource.data.criadoEm == resource.data.criadoEm;
      allow delete: if licAdmin() || autor24h();
    }
    match /licoesCategorias/{id} {
      allow read: if true;
      allow create: if sessao() && id.size() <= 60 && request.resource.data.autorUid == request.auth.uid
                    && request.resource.data.criadoEm == request.time
                    && request.resource.data.nome is string && request.resource.data.nome.size() >= 3 && request.resource.data.nome.size() <= 40;
      allow update, delete: if licAdmin();
    }
    match /licoesConfig/{doc}   { allow read: if true; allow write: if licAdmin(); }
    match /licoesAdmins/{email} {
      allow read: if licAdmin() || (logado() && request.auth.token.email == email);
      allow write: if licAdmin();
    }

    // ---------- Modelos de lançamento UAU (sem login) ----------
    match /modelosCadastro/{id} {
      allow read: if true;
      allow create, update: if
        request.resource.data.keys().hasOnly(['tipo', 'empresa', 'nome', 'cnpj', 'texto', 'unid', 'insumo', 'preco',
          'numContrato', 'fornecedor', 'objeto', 'valorContrato', 'itens', 'medicoes', 'atualizadoEm'])
        && request.resource.data.tipo in ['fornecedor', 'item', 'planejamento', 'contrato']
        && txt(request.resource.data, 'nome', 200) && txt(request.resource.data, 'texto', 300)
        && txt(request.resource.data, 'objeto', 600) && txt(request.resource.data, 'cnpj', 30)
        && (!('itens' in request.resource.data) || request.resource.data.itens.size() < 200);
      allow delete: if editor();
    }
  }
}
